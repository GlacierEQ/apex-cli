from __future__ import annotations

import argparse
import hashlib
import json
from pathlib import Path


def sha256(path: Path) -> str:
    h = hashlib.sha256()
    with path.open("rb") as handle:
        for chunk in iter(lambda: handle.read(1024 * 1024), b""):
            h.update(chunk)
    return h.hexdigest()


def main() -> int:
    parser = argparse.ArgumentParser()
    parser.add_argument("--root", required=True)
    parser.add_argument("--manifest", default="RUNTIME_PROJECTION.json")
    parser.add_argument("--expect-commit")
    args = parser.parse_args()
    root = Path(args.root).resolve()
    manifest_path = root / args.manifest
    if not manifest_path.is_file():
        print(json.dumps({"status": "FAILED", "errors": [f"missing {manifest_path}"]}, indent=2))
        return 1
    manifest = json.loads(manifest_path.read_text(encoding="utf-8"))
    errors = []
    if manifest.get("schema") != "glaciereq.runtime-projection.v2":
        errors.append("projection schema drift")
    if manifest.get("canonical_repository") != "GlacierEQ/apex-boot-core":
        errors.append("canonical repository drift")
    if args.expect_commit and manifest.get("canonical_commit") != args.expect_commit:
        errors.append("canonical commit drift")
    for rel, expected in manifest.get("files", {}).items():
        path = root / rel
        if not path.is_file():
            errors.append(f"missing projected file: {rel}")
            continue
        actual = sha256(path)
        if actual != expected:
            errors.append(f"checksum drift: {rel}")
    print(json.dumps({
